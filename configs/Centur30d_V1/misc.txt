####################################################################################
#				 twospeed.comp
####################################################################################

component twospeed "Checks if spindle speed is within range";
pin in float speed_in "Current spindle speed from motion";
pin in float ref_speed "Minimum speed threshold";
pin out bit is_on "True if speed_in > ref_speed";
function _;
license "GPL";
;;
FUNCTION(_) {
    if (speed_in > ref_speed) {
        is_on = 1;		//Activate low-speed contactor
    } else {
        is_on = 0;		//Activate high-speed contactor
    }
}





####################################################################################
#				   spindle_check.hal
####################################################################################


#If S < 1200 → LowSpeed (contactor KM2)
#If S ≥ 1200 → HighSpeed (contactor KM3)


loadrt twospeed count=1
addf twospeed.0 servo-thread

# Set threshold (e.g., if S is greater than 1200)
setp twospeed.0.ref_speed 1200

# Connect the spindle speed to the checker
net speed-raw motion.spindle-speed-out => twospeed.0.speed-in
